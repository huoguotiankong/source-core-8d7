var QF_OfficialSeverity=(function(){
    function n(v){v=Number(v||0);return isFinite(v)?v:0;}
    function text(v){try{return String(v==null?"":v);}catch(_e){return "";}}
    function reviewDebt(endpoint,lastError,lastCode){
        var ep=text(endpoint).toLowerCase(),er=(text(lastError)+" "+text(lastCode)).toLowerCase();
        var reviewPath=/v1\/chapterreview\/(?:getparagraphshotcomments|getchapterendreview)/.test(ep);
        var reviewScope=/^(?:review-reader:|argus:)/.test(ep);
        var sign=/auth\/sign|qdsign|qdinfo|signature|sign error|signature_error|验签|签名/.test(er);
        return reviewPath&&reviewScope&&sign;
    }
    function eggFallbackOk(ctx){
        try{
            if(!ctx||typeof QF_EggReport==="undefined"||!QF_EggReport||typeof QF_EggReport.load!=="function")return false;
            var r=QF_EggReport.load(ctx),o=r&&r.output||{},age=Date.now()-n(r&&r.ts);
            return !!(r&&age>=0&&age<=10*60*1000&&n(o.images)>0&&text(o.source));
        }catch(_e){return false;}
    }
    function optional(endpoint,lastError,catalogOk,eggOk){
        var ep=text(endpoint).toLowerCase(),er=text(lastError).toLowerCase();
        if(/freshman\/limitfreepopup/.test(ep)&&/-11205\b/.test(er))return "freshman-limitfree";
        if(/bookshelf\/gethighlevelbookreadinfo/.test(ep)&&/-301\b/.test(er))return "highlevel-readinfo";
        if(catalogOk&&/chapterlist\/chapterlist/.test(ep)&&/-1004\b/.test(er))return "catalog-v3-probe";
        if(eggOk&&/midpage\/pageinfo/.test(ep)&&/-280015\b/.test(er))return "egg-pageinfo-fallback-success";
        return "";
    }
    function classifyEndpoint(x,catalog,eggOk){
        x=x||{};var errors=n(x.errors);if(errors<=0)return {level:"ok",count:0,reason:""};
        var co=catalog&&catalog.output?catalog.output:{},catalogOk=n(co.chapters)>0;
        if(reviewDebt(x.endpoint,x.lastError,x.lastCode))return {level:"debt",count:errors,reason:"official-review-sign"};
        var op=optional(x.endpoint,x.lastError,catalogOk,!!eggOk);if(op)return {level:"optional",count:errors,reason:op};
        return {level:"actionable",count:errors,reason:"unclassified"};
    }
    function summary(official,catalog,ctx){
        official=official||{};var eps=Array.isArray(official.endpoints)?official.endpoints:[],
            eggOk=eggFallbackOk(ctx),
            out={raw:n(official.errors),actionable:0,debt:0,optional:0,classified:0,items:[],eggFallbackOk:eggOk};
        for(var i=0;i<eps.length;i++){
            var x=eps[i]||{},c=classifyEndpoint(x,catalog,eggOk),cnt=n(c.count);
            if(cnt<=0)continue;
            if(c.level==="actionable")out.actionable+=cnt;
            else if(c.level==="debt")out.debt+=cnt;
            else if(c.level==="optional")out.optional+=cnt;
            out.classified+=cnt;
            out.items.push({endpoint:text(x.endpoint),errors:cnt,level:c.level,reason:c.reason,lastError:text(x.lastError),lastCode:text(x.lastCode)});
        }
        /* summary endpoints 最多16项；若 raw 比已分类更多，剩余一律按 actionable，禁止静默吞错。 */
        if(out.raw>out.classified)out.actionable+=(out.raw-out.classified);
        return out;
    }
    function recordLevel(x,catalog,officialSeverity,ctx){
        x=x||{};var area=text(x.area),name=text(x.name),cap=text(x.cap),detail=text(x.detail),code=text(x.code);
        if(area!=="official-request")return "actionable";
        var endpoint=name+":"+cap;
        if(reviewDebt(endpoint,detail,code))return "debt";
        if(/freshman\/limitfreepopup/i.test(cap)&&/-11205\b/.test(detail+" "+code))return "optional";
        if(/bookshelf\/gethighlevelbookreadinfo/i.test(cap)&&/-301\b/.test(detail+" "+code))return "optional";

        var co=catalog&&catalog.output?catalog.output:{},catalogOk=n(co.chapters)>0,
            sev=officialSeverity||{},is1004=/-1004\b/.test((detail+" "+code).toLowerCase());
        /* 优先使用完整 cap；匿名 app 记录只在 endpoint 汇总已证明“无 actionable 且存在 optional”时降级。 */
        if(catalogOk&&is1004&&/chapterlist\/chapterlist/i.test(cap))return "optional";
        if(catalogOk&&is1004&&/^app$/i.test(name)&&n(sev.actionable)===0&&n(sev.optional)>0)return "optional";
        if(eggFallbackOk(ctx)&&/-280015\b/.test((detail+" "+code).toLowerCase())&&/midpage\/pageinfo/i.test(cap))return "optional";
        return "actionable";
    }
    function tag(level){
        return level==="debt"?"KNOWN-DEBT":(level==="optional"?"OPTIONAL":"ACTIONABLE");
    }
    return {summary:summary,classifyEndpoint:classifyEndpoint,recordLevel:recordLevel,tag:tag,eggFallbackOk:eggFallbackOk};
})();
