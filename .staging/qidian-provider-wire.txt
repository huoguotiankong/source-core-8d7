/* ===== 起点发现页：沿用成熟起点书源的轻量 CSRF 机制 =====
 * 发现 URL 中通过 {{generateCsrfToken()}} 在真正发起请求时调用。
 * 不要求用户手动刷新 Cookie；本地没有 token 时，只静默读取一次起点 document.cookie。
 */
function generateCsrfToken() {
    var qd = "https://m.qidian.com";
    var token = "";
    try {
        var current = String(this.cookie.getCookie(qd) || "");
        var m = current.match(/_csrfToken=([^;]*)/);
        token = String(m && m[1] ? m[1] : "");
    } catch(e0) {}

    try {
        if (!token) {
            try { this.cookie.removeCookie(qd); } catch(e1) {}
            var webCookie = this.java.webView(null, qd, "document.cookie");
            if (webCookie) {
                try { this.cookie.setCookie(qd, String(webCookie)); } catch(e2) {}
            }
            var refreshed = "";
            try { refreshed = String(this.cookie.getCookie(qd) || ""); } catch(e3) {}
            var m2 = refreshed.match(/_csrfToken=([^;]*)/);
            token = String(m2 && m2[1] ? m2[1] : "");
        }
    } catch(e4) {}

    return token;
}
