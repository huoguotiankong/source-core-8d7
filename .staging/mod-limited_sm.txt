function qfSmAbsV30(base,u){
    u=String(u||"").trim();
    if(!u)return "";
    if(/^https?:\/\//i.test(u))return u;
    if(/^\/\//.test(u))return "https:"+u;
    base=qfSmNormBaseV30(base);
    var m=base.match(/^(https?:\/\/[^\/]+)/i);
    var origin=m?m[1]:base;
    if(u.charAt(0)==="/")return origin+u;
    return base+"/"+u.replace(/^\/+/,"");
}
